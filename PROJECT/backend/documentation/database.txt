-- 0. UTILITY FUNCTION

CREATE OR REPLACE FUNCTION handle_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 17a. Sync course completion → roadmap step status
-- When a user completes a course (or hits 100%), mark the
-- matching roadmap step as completed automatically.

CREATE OR REPLACE FUNCTION sync_course_to_roadmap_step()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.status = 'completed' OR NEW.progress_pct = 100 THEN
        UPDATE public.ai_roadmap_steps
        SET
            status       = 'completed',
            completed_at = NOW(),
            updated_at   = NOW()
        WHERE
            course_id  = NEW.course_id
            AND profile_id = NEW.profile_id
            AND status NOT IN ('completed', 'skipped');   -- don't overwrite already-done steps
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_sync_course_progress
AFTER UPDATE ON public.user_progress
FOR EACH ROW
EXECUTE FUNCTION sync_course_to_roadmap_step();


-- 17b. Auto-unlock next roadmap step when current one completes

CREATE OR REPLACE FUNCTION unlock_next_roadmap_step()
RETURNS TRIGGER AS $$
DECLARE
    next_order integer;
BEGIN
    IF NEW.status = 'completed' AND OLD.status <> 'completed' THEN
        -- Find the next step_order in the same roadmap
        SELECT step_order INTO next_order
        FROM public.ai_roadmap_steps
        WHERE roadmap_id = NEW.roadmap_id
          AND step_order > NEW.step_order
          AND status = 'locked'
        ORDER BY step_order ASC
        LIMIT 1;

        IF next_order IS NOT NULL THEN
            UPDATE public.ai_roadmap_steps
            SET status     = 'available',
                updated_at = NOW()
            WHERE roadmap_id  = NEW.roadmap_id
              AND step_order   = next_order;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_unlock_next_step
AFTER UPDATE ON public.ai_roadmap_steps
FOR EACH ROW
EXECUTE FUNCTION unlock_next_roadmap_step();


-- 17c. Auto-update ai_roadmaps.status when all steps complete

CREATE OR REPLACE FUNCTION check_roadmap_completion()
RETURNS TRIGGER AS $$
DECLARE
    total_steps     integer;
    completed_steps integer;
BEGIN
    IF NEW.status = 'completed' AND OLD.status <> 'completed' THEN
        SELECT COUNT(*) INTO total_steps
        FROM public.ai_roadmap_steps
        WHERE roadmap_id = NEW.roadmap_id;

        SELECT COUNT(*) INTO completed_steps
        FROM public.ai_roadmap_steps
        WHERE roadmap_id = NEW.roadmap_id
          AND status = 'completed';

        IF total_steps > 0 AND total_steps = completed_steps THEN
            UPDATE public.ai_roadmaps
            SET status     = 'completed',
                updated_at = NOW()
            WHERE id = NEW.roadmap_id;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_roadmap_completion
AFTER UPDATE ON public.ai_roadmap_steps
FOR EACH ROW
EXECUTE FUNCTION check_roadmap_completion();
